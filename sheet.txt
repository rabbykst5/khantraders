function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const action = e.parameter.action || '';

  // ১. ক্যাটাগরি লোড
  if (action === 'getCategories') {
    const prodSheet = ss.getSheetByName('Products');
    if (!prodSheet) {
      return ContentService.createTextOutput(JSON.stringify([]))
        .setMimeType(ContentService.MimeType.JSON);
    }
    const rows = prodSheet.getDataRange().getValues();
    const categoriesMap = new Map();

    for (let i = 1; i < rows.length; i++) {
      const catKey = String(rows[i][5] || '').trim();
      const catName = String(rows[i][6] || '').trim();
      if (catKey && catName && !categoriesMap.has(catKey)) {
        categoriesMap.set(catKey, catName);
      }
    }

    const categories = [];
    categoriesMap.forEach((name, key) => {
      categories.push({ key: key, name: name });
    });

    return ContentService.createTextOutput(JSON.stringify(categories))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // ২. প্রোডাক্ট লোড
  if (action === 'getProducts') {
    const prodSheet = ss.getSheetByName('Products');
    if (!prodSheet) {
      return ContentService.createTextOutput(JSON.stringify([]))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const rows = prodSheet.getDataRange().getValues();
    const products = [];

    for (let i = 1; i < rows.length; i++) {
      const r = rows[i];
      const prodId = String(r[0] || '').trim();
      if (!prodId) continue;

      let imgUrl = String(r[11] || '').trim();
      if (imgUrl.includes('drive.google.com')) {
        const match = imgUrl.match(/\/d\/([a-zA-Z0-9_-]+)/) || imgUrl.match(/id=([a-zA-Z0-9_-]+)/);
        if (match && match[1]) {
          imgUrl = 'https://lh3.googleusercontent.com/d/' + match[1];
        }
      }

      let rawSpecs = String(r[10] || '').trim();
      let specsArray = rawSpecs ? rawSpecs.split(/[,،]+/).map(s => s.trim()).filter(s => s.length > 0) : [];
      let stockVal = r[8];
      let isStockTrue = stockVal === true || String(stockVal).trim().toUpperCase() === 'TRUE';

      let badgeVal = String(r[7] || '').trim();
      if (badgeVal.toUpperCase() === 'TRUE' || badgeVal.toUpperCase() === 'FALSE') {
        badgeVal = '';
      }

      products.push({
        id: prodId,
        name: String(r[1] || '').trim(),
        price: Number(r[2]) || 0,
        oldPrice: Number(r[3]) || 0,
        unit: String(r[4] || '').trim(),
        category: String(r[5] || 'all').trim(),
        categoryName: String(r[6] || '').trim(),
        badge: badgeVal,
        stock: isStockTrue,
        description: String(r[9] || '').trim(),
        specs: specsArray,
        image: imgUrl,
        images: imgUrl ? [imgUrl] : []
      });
    }

    return ContentService.createTextOutput(JSON.stringify(products))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // ৩. অর্ডার লিস্ট লোড
  if (action === 'getOrders') {
    const sheet = ss.getActiveSheet();
    const rows = sheet.getDataRange().getValues();
    const orders = [];

    for (let i = 1; i < rows.length; i++) {
      const row = rows[i];
      if (!row[1]) continue;
      
      orders.push({
        date: row[0],
        orderId: row[1],
        name: row[2],
        phone: row[3],
        address: row[4],
        thana: row[5],
        district: row[6],
        item: row[7],
        total: row[8],
        payMethod: row[9],
        trxId: row[10],
        status: row[11] || 'পেন্ডিং'
      });
    }

    return ContentService.createTextOutput(JSON.stringify(orders.reverse()))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // ৪. অ্যাডমিন থেকে অর্ডার স্ট্যাটাস আপডেট করার ফিক্সড হ্যান্ডলার
  if (action === 'updateOrderStatus') {
    const orderId = String(e.parameter.orderId || '').trim();
    const newStatus = String(e.parameter.status || '').trim();

    if (!orderId || !newStatus) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Missing parameters" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const sheet = ss.getActiveSheet();
    const rows = sheet.getDataRange().getValues();
    let targetRowIndex = -1;

    for (let i = 1; i < rows.length; i++) {
      if (String(rows[i][1]).trim().toLowerCase() === orderId.toLowerCase()) {
        targetRowIndex = i + 1;
        break;
      }
    }

    if (targetRowIndex === -1) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Order ID not found" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // Col L (12th column) এ স্ট্যাটাস সেট করা
    sheet.getRange(targetRowIndex, 12).setValue(newStatus);

    return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Status updated successfully" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // ৫. ট্র্যাকিং সার্চ
  const sheet = ss.getActiveSheet();
  const rows = sheet.getDataRange().getValues();
  const query = e.parameter.id ? e.parameter.id.trim() : (e.parameter.phone ? e.parameter.phone.trim() : '');
  
  if (!query) {
    return ContentService.createTextOutput(JSON.stringify({ found: false, message: "Query missing" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const cleanQuery = query.replace(/[^0-9]/g, '');
  const isPhoneSearch = cleanQuery.length >= 10;
  const targetId = query.toUpperCase();
  const matchingOrders = [];

  for (let i = 1; i < rows.length; i++) {
    const row = rows[i];
    const currentId = String(row[1] || '').trim().toUpperCase(); 
    let currentPhone = String(row[3] || '').replace(/[^0-9]/g, '');
    if (currentPhone.startsWith('880')) currentPhone = currentPhone.slice(3);
    else if (currentPhone.startsWith('0')) currentPhone = currentPhone.slice(1);

    let match = false;
    if (isPhoneSearch) {
      let qPhone = cleanQuery;
      if (qPhone.startsWith('880')) qPhone = qPhone.slice(3);
      else if (qPhone.startsWith('0')) qPhone = qPhone.slice(1);
      if (currentPhone === qPhone) match = true;
    } else {
      if (currentId === targetId) match = true;
    }
    
    if (match) {
      matchingOrders.push({
        date: row[0], orderId: row[1], name: row[2], phone: row[3],
        address: row[4], thana: row[5], district: row[6], item: row[7],
        total: row[8], payMethod: row[9], trxId: row[10],
        status: row[11] || 'পেন্ডিং', courierCode: row[12] ? String(row[12]).trim() : ''
      });
    }
  }

  if (matchingOrders.length > 0) {
    return ContentService.createTextOutput(JSON.stringify({ found: true, orders: matchingOrders }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  return ContentService.createTextOutput(JSON.stringify({ found: false, message: "No orders found" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let data;
    
    if (e.postData && e.postData.contents) {
      data = JSON.parse(e.postData.contents);
    } else {
      data = e.parameter;
    }

    let prodSheet = ss.getSheetByName('Products');
    if (!prodSheet) {
      prodSheet = ss.insertSheet('Products');
    }

    if (data.action === 'addProduct') {
      const newProdRow = [
        String(data.id || '').trim(),
        String(data.name || '').trim(),
        Number(data.price) || 0,
        Number(data.oldPrice) || 0,
        String(data.unit || '').trim(),
        String(data.category || '').trim(),
        String(data.categoryName || '').trim(),
        String(data.badge || '').trim(),
        true,
        String(data.description || '').trim(),
        String(data.specs || '').trim(),
        String(data.image || '').trim()
      ];
      prodSheet.appendRow(newProdRow);
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    if (data.action === 'updateProduct') {
      const rows = prodSheet.getDataRange().getValues();
      let targetRowIndex = -1;

      for (let i = 1; i < rows.length; i++) {
        if (String(rows[i][0]).trim().toLowerCase() === String(data.id).trim().toLowerCase()) {
          targetRowIndex = i + 1;
          break;
        }
      }

      if (targetRowIndex === -1) {
        return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Product ID not found" }))
          .setMimeType(ContentService.MimeType.JSON);
      }

      prodSheet.getRange(targetRowIndex, 2).setValue(String(data.name || '').trim());
      prodSheet.getRange(targetRowIndex, 3).setValue(Number(data.price) || 0);
      prodSheet.getRange(targetRowIndex, 4).setValue(Number(data.oldPrice) || 0);
      prodSheet.getRange(targetRowIndex, 5).setValue(String(data.unit || '').trim());
      prodSheet.getRange(targetRowIndex, 6).setValue(String(data.category || '').trim());
      prodSheet.getRange(targetRowIndex, 7).setValue(String(data.categoryName || '').trim());
      prodSheet.getRange(targetRowIndex, 8).setValue(String(data.badge || '').trim());
      prodSheet.getRange(targetRowIndex, 9).setValue(data.stock === true || String(data.stock).toUpperCase() === 'TRUE');
      prodSheet.getRange(targetRowIndex, 11).setValue(String(data.specs || '').trim());
      prodSheet.getRange(targetRowIndex, 12).setValue(String(data.image || '').trim());

      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ success: true }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

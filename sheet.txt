// ১. রিড এন্ডপয়েন্ট (doGet) — ট্র্যাকিং এবং প্রোডাক্ট লোড
// =========================================================
function doGet(e) {
  // ক. ওয়েবসাইট থেকে প্রোডাক্ট লোড করার রিকোয়েস্ট (action === 'getProducts')
  if (e.parameter.action === 'getProducts') {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const prodSheet = ss.getSheetByName('Products');
    if (!prodSheet) {
      return ContentService.createTextOutput(JSON.stringify([]))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const rows = prodSheet.getDataRange().getValues();
    const products = [];

    // Row 1 বাদ দিয়ে Row 2 থেকে প্রোডাক্ট পড়া শুরু হবে
    for (let i = 1; i < rows.length; i++) {
      const r = rows[i];
      if (!r[0]) continue; // ID না থাকলে স্কিপ করবে

      // গুগল ড্রাইভ লিংক অটো কনভার্টার লজিক
      let imgUrl = String(r[7] || '').trim();
      if (imgUrl.includes('drive.google.com')) {
        // লিংক থেকে File ID বের করা (/d/ID অথবা id=ID)
        const match = imgUrl.match(/\/d\/([a-zA-Z0-9_-]+)/) || imgUrl.match(/id=([a-zA-Z0-9_-]+)/);
        if (match && match[1]) {
          imgUrl = 'https://lh3.googleusercontent.com/d/' + match[1];
        }
      }

      products.push({
        id: String(r[0]).trim(),
        name: String(r[1] || '').trim(),
        price: Number(r[2]) || 0,
        oldPrice: Number(r[3]) || 0,
        unit: String(r[4] || '').trim(),
        category: String(r[5] || 'all').trim(),
        categoryName: String(r[6] || '').trim(),
        images: imgUrl ? [imgUrl] : [],
        badge: String(r[8] || '').trim(),
        stock: r[9] === true || String(r[9]).trim().toUpperCase() === 'TRUE'
      });
    }

    return ContentService.createTextOutput(JSON.stringify(products))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // খ. ট্র্যাকিং পেজের জন্য রিকোয়েস্ট (id বা phone দিয়ে সার্চ)
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  const rows = sheet.getDataRange().getValues();
  const query = e.parameter.id ? e.parameter.id.trim() : (e.parameter.phone ? e.parameter.phone.trim() : '');
  
  if (!query) {
    return ContentService.createTextOutput(JSON.stringify({ found: false, message: "Query missing" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  const cleanQuery = query.replace(/[^0-9]/g, '');
  const isPhoneSearch = cleanQuery.length >= 10;
  const targetId = query.toUpperCase();

  const matchingOrders = [];

  for (let i = 1; i < rows.length; i++) {
    const row = rows[i];
    const currentId = String(row[1] || '').trim().toUpperCase(); 
    let currentPhone = String(row[3] || '').replace(/[^0-9]/g, '');
    if (currentPhone.startsWith('880')) currentPhone = currentPhone.slice(3);
    else if (currentPhone.startsWith('0')) currentPhone = currentPhone.slice(1);

    let match = false;
    if (isPhoneSearch) {
      let qPhone = cleanQuery;
      if (qPhone.startsWith('880')) qPhone = qPhone.slice(3);
      else if (qPhone.startsWith('0')) qPhone = qPhone.slice(1);
      if (currentPhone === qPhone) match = true;
    } else {
      if (currentId === targetId) match = true;
    }
    
    if (match) {
      matchingOrders.push({
        date: row[0],
        orderId: row[1],
        name: row[2],
        phone: row[3],
        address: row[4],
        thana: row[5],
        district: row[6],
        item: row[7],
        total: row[8],
        payMethod: row[9],
        trxId: row[10],
        status: row[11] || 'পেন্ডিং',
        courierCode: row[12] ? String(row[12]).trim() : ''
      });
    }
  }

  if (matchingOrders.length > 0) {
    return ContentService.createTextOutput(JSON.stringify({ found: true, orders: matchingOrders }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  return ContentService.createTextOutput(JSON.stringify({ found: false, message: "No orders found" }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ২. অর্ডার সেভ করার এন্ডপয়েন্ট (সবার উপরে রো ২-তে ইনসার্ট)
// =========================================================
function doPost(e) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    let data;
    
    if (e.postData && e.postData.contents) {
      data = JSON.parse(e.postData.contents);
    } else {
      data = e.parameter;
    }

    const timestamp = Utilities.formatDate(new Date(), "Asia/Dhaka", "yyyy-MM-dd hh:mm a");
    const orderId = data.orderId || '';
    const name = data.name || '';
    const phone = data.phone || '';
    const address = data.address || '';
    const thana = data.thana || '';
    const district = data.district || '';
    const item = data.items || '';
    const total = data.total || 0;
    
    let payMethod = 'ক্যাশ অন ডেলিভারি';
    let trxId = 'N/A';

    if (data.payMethod && (data.payMethod.includes('bKash') || data.payMethod.includes('mfs'))) {
      payMethod = 'বিকাশ / নগদ';
      if (data.payMethod.includes('Trx:')) {
        trxId = data.payMethod.split('Trx:')[1].replace(')', '').trim();
      } else if (data.trxId) {
        trxId = data.trxId.trim();
      }
    } else if (data.trxId && data.trxId.trim().length > 0) {
      payMethod = 'বিকাশ / নগদ';
      trxId = data.trxId.trim();
    }
    
    const status = 'পেন্ডিং';

    const newRowData = [
      timestamp,
      orderId,
      name,
      phone,
      address,
      thana,
      district,
      item,
      total,
      payMethod,
      trxId,
      status,
      '' // Courier Code কলাম
    ];

    sheet.insertRowBefore(2);
    sheet.getRange(2, 1, 1, newRowData.length).setValues([newRowData]);

    const orderData = {
      orderId: orderId,
      name: name,
      phone: phone,
      district: district,
      thana: thana,
      address: address,
      items: item,
      total: total,
      payMethod: payMethod,
      trxId: trxId,
      time: timestamp
    };

    // ১. অ্যাডমিনের ইমেইল অ্যালার্ট
    sendAdminEmail(orderData);

    // ২. UltraMsg দিয়ে গ্রাহকের WhatsApp মেসেজ
    sendCustomerWhatsApp(orderData);

    return ContentService.createTextOutput(JSON.stringify({ success: true }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

      `💰 *পরিশোধযোগ্য বিল:* ৳ ${Number(order.total).toLocaleString('en-BD')}\n` +
      `💳 *পেমেন্ট পদ্ধতি:* ${order.payMethod}\n` +
      (order.trxId !== 'N/A' ? `📝 *TrxID:* ${order.trxId}\n` : '') +

    const payload = {
      token: ULTRAMSG_CONFIG.TOKEN,
      to: formattedPhone,
      body: messageBody
    };

    const options = {
      method: "POST",
      contentType: "application/x-www-form-urlencoded",
      payload: payload,
      muteHttpExceptions: true
    };

    const url = `https://api.ultramsg.com/${ULTRAMSG_CONFIG.INSTANCE_ID}/messages/chat`;
    const res = UrlFetchApp.fetch(url, options);
    Logger.log("UltraMsg WhatsApp Response: " + res.getContentText());
